% ---- entry: org (left) | dates (right), then role ----
\newcommand{\entry}[3]{%
	\noindent%
	\begin{minipage}[t]{0.72\textwidth}
		\textbf{#1}
	\end{minipage}%
	\begin{minipage}[t]{0.26\textwidth}
		\raggedleft\textbf{#2}
	\end{minipage}\par%
	\noindent\textit{#3}\par%
	\vspace{0.04em}%
}

% ---- one-line entry: left | right (no role line) ----
\newcommand{\entryline}[2]{%
	\noindent%
	\begin{minipage}[t]{0.72\textwidth}
		\textbf{#1}
	\end{minipage}%
	\begin{minipage}[t]{0.26\textwidth}
		\raggedleft #2
	\end{minipage}\par%
	\vspace{0.04em}%
}

% ---- two-line cert entry: title (bold), provider + date below ----
\newcommand{\entrycert}[3]{%
	\noindent\textbf{#1}\par%
	\noindent%
	\begin{minipage}[t]{0.72\textwidth}
		#2
	\end{minipage}%
	\begin{minipage}[t]{0.26\textwidth}
		\raggedleft #3
	\end{minipage}\par%
	\vspace{0.03em}%
}

% ---- bullet item ----

}
